## 公開 API

```rust
// oidc.rs
pub fn authorize_url(client_id: &str, redirect_uri: &str, state: &str, nonce: &str) -> String;
#[derive(Debug, PartialEq)]
pub struct IdClaims { pub sub: String, pub name: Option<String> }
#[derive(Debug, PartialEq)]
pub enum OidcError { Malformed, Issuer, Audience, Nonce, Expired, Http(String) }
pub fn parse_id_token(jwt: &str, client_id: &str, nonce: &str, now_unix: i64) -> Result<IdClaims, OidcError>;

#[async_trait]
pub trait OidcHttp: Send + Sync {
    async fn token(&self, code: &str, client_id: &str, client_secret: &str, redirect_uri: &str) -> Result<String /* id_token */, OidcError>;
}
pub struct HttpOidc; // reqwest, not unit-tested

// session.rs
pub struct Sessions; // pending: HashMap<state, (nonce, expires)>, cookies: HashMap<sid, (user, expires)>
impl Sessions {
    pub fn begin(&mut self, now: i64) -> (String /*state*/, String /*nonce*/); // 10 min
    pub fn take_pending(&mut self, state: &str, now: i64) -> Option<String /*nonce*/>; // once
    pub fn create_cookie(&mut self, user: &str, now: i64) -> String /*sid*/;          // 24 h
    pub fn user_for(&self, sid: &str, now: i64) -> Option<String>;
}
pub fn random_token() -> String; // 32 bytes base64url no pad
```
