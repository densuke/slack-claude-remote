pub fn parse_verdict(text: &str) -> Option<(String /*request_id*/, protocol::Behavior)>;
pub fn prompt_text(tool_name: &str, description: &str, input_preview: &str, request_id: &str) -> String;

## 組み込み

- agent から `PermissionRequest` を受けたら、その agent 名に束縛されたスレッドを bindings から探す。
  複数あれば最後に束縛したもの。無ければ捨てる。pending に登録し、`prompt_text` をスレッドに投稿
  - bindings に時刻が無いので、`Binding` に `created: i64` を追加して状態ファイル互換を保つ（`#[serde(default)]`）
- スレッド返信の処理で、Inbound 送信の前に `parse_verdict` を試す。
  一致し、pending にあり、chat_id が一致し、投稿者が binding.user なら `PermissionVerdict` を送り、pending から消し、Inbound は送らない。
  それ以外は通常の Inbound
- 受信のたびに `sweep`
